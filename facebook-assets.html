Xuất file PNG

Cách nhanh nhất: bấm Mở 1:1 ↗ rồi chụp màn hình. Xuất hàng loạt bằng Chrome ở chế độ nền:

CH="/Applications/Google Chrome.app/Contents/MacOS/Google Chrome"
DIR="$PWD"; mkdir -p img/fb
printf '%s\n' "cv-center 1640 624" "cv-left 1640 624" "cv-stats 1640 624" \
  "pf-solid 1080 1080" "pf-dark 1080 1080" "pf-mono 1080 1080" \
  "po-hero 1200 1200" "po-person 1200 1200" "po-stats 1200 1200" \
  "po-agenda 1200 1200" "po-benefit 1200 1200" "po-cta 1200 1200" \
  "og-main 1200 630" |
while read id w h; do
  "$CH" --headless --disable-gpu --hide-scrollbars --force-device-scale-factor=1 \
        --virtual-time-budget=6000 --window-size=$w,$h \
        --screenshot="img/fb/$id.png" "file://$DIR/facebook-assets.html?only=$id"
done
