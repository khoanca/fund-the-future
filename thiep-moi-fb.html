Xuất file PNG

Bấm Mở 1:1 ↗ rồi chụp màn hình, hoặc xuất hàng loạt bằng Chrome ở chế độ nền (thêm &clean=1 vào URL nếu muốn nền trống để ghép ảnh ngoài):

CH="/Applications/Google Chrome.app/Contents/MacOS/Google Chrome"
DIR="$PWD"; mkdir -p img/fb
printf '%s\n' "tm-doc 1080 1350" "tm-vuong 1200 1200" "tm-ngang 1920 1005" |
while read id w h; do
  "$CH" --headless --disable-gpu --hide-scrollbars --force-device-scale-factor=1 \
        --virtual-time-budget=6000 --window-size=$w,$h \
        --screenshot="img/fb/$id.png" "file://$DIR/thiep-moi-fb.html?only=$id"
done
